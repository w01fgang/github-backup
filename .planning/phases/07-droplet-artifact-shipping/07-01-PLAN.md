```typescript
function group1SyncOneRepoContract(cfg: Config, dropInfo: DropletInfo, target: { source: string; owner: string; repo: string }): void {
  console.log("\n— Group 1: sync-one-repo.sh ships + per-repo contract (SC#1, D-05) —");
  const { ip } = dropInfo; const user = cfg.sshUser; const key = cfg.sshKeyPath;

  // 1a. ships + executable
  assert(sshExitsZero(ip, user, key, `test -x ${REMOTE_SYNC_ONE_REPO}`),
    `${REMOTE_SYNC_ONE_REPO} exists and is executable on droplet`);

  // 1b. functional one-repo invocation. Source backup.env first so GITHUB_TOKEN
  //     is in env, then exec sync-one-repo.sh against the pre-arranged target.
  //     Capture stdout+stderr so we can assert on RESULT_TAG appearance in log.
  const { source, owner, repo } = target;
  // Defence-in-depth slug shape (already validated by config loader, but verify
  // interpolates into a single-quoted ssh payload — re-check here per threat model).
  const SLUG_RE = /^[A-Za-z0-9._-]+$/;
  if (!SLUG_RE.test(source) || !SLUG_RE.test(owner) || !SLUG_RE.test(repo)) {
    bail(`Phase 7 verify: target slug "${source}/${owner}/${repo}" has shell-unsafe chars; refusing`);
  }

  const invokeCmd =
    `set -a; source ${REMOTE_ENV(cfg)}; set +a; ` +
    `${REMOTE_SYNC_ONE_REPO} ${source} ${owner} ${repo} >/dev/null 2>&1; ` +
    `echo exit=$?`;
  const out = sshCapture(ip, user, key, invokeCmd);
  assert(/exit=0/.test(out),
    `sync-one-repo.sh ${source} ${owner} ${repo} exited 0`);

  // 1c. namespaced mirror dir present (D-07 layout)
  const mirrorPath = `${REMOTE_DIR}/${source}/${owner}_${repo}.git`;
  assert(sshExitsZero(ip, user, key, `test -d ${mirrorPath}`),
    `namespaced mirror dir ${mirrorPath} exists (D-07)`);

  // 1d. RESULT_TAG log line present for this owner/repo (D-15 contract)
  const grepCmd = `grep -F "${RESULT_TAG} source=${source} owner=${owner} repo=${repo}" ${REMOTE_LOG} | tail -1`;
  const tag = sshCapture(ip, user, key, grepCmd);
  assert(tag.length > 0 && /action=(clone|update)/.test(tag),
    `${REMOTE_LOG} contains ${RESULT_TAG} for ${source}/${owner}/${repo} with action=clone|update`);
}
```

```typescript
function group4EndToEnd(cfg: Config, dropInfo: DropletInfo, target: { source: string; owner: string; repo: string }): void {
  console.log("\n— Group 4: github-backup.sh end-to-end (SC#4, D-08) —");
  const { ip } = dropInfo; const user = cfg.sshUser; const key = cfg.sshKeyPath;
  const { source, owner, repo } = target;

    sshCapture(ip, user, key, `wc -c < ${REMOTE_LOG} 2>/dev/null || echo 0`).trim(),
    10
  ) || 0;
  info(`log size before: ${sizeBefore} bytes`);

  // 4b. run cron path. github-backup.sh is the orchestrator — it iterates all
  // sources + repos through filter_repos and dispatches sync-one-repo.sh. The

  const runCmd = `${REMOTE_BACKUP_SH} >/dev/null 2>&1; echo exit=$?`;

  assert(sshExitsZero(ip, user, key, `test -d ${mirrorPath}`),
```

**Target selection helper:**
```typescript
const SLUG_RE = /^[A-Za-z0-9._-]+$/;
function parseRepoSlug(slug: string): { owner: string; repo: string } | null {
  const m = slug.match(/^([A-Za-z0-9._-]+)\/([A-Za-z0-9._-]+)$/);
  return m ? { owner: m[1], repo: m[2] } : null;
}

function passesFilter(s: NormalizedSource, owner: string, repo: string): boolean {
  // Mirror filter_repos.sh semantics in TS for local pre-selection. Deny wins.
  const full = `${owner}/${repo}`;
  const name = repo;
  const matchAny = (pats: string[]) => pats.some((p) => {
    if (p.includes("/")) return globMatch(p, full);
    return globMatch(p, name);
  });
  if (s.deny.length > 0 && matchAny(s.deny)) return false;
  if (s.allow.length === 0) return true;
  return matchAny(s.allow);
}
// Tiny bash-style glob (* ? only — no [..] needed for v1.1 patterns).
function globMatch(pat: string, s: string): boolean {
  const rx = "^" + pat.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".") + "$";
  return new RegExp(rx).test(s);
}

function chooseTarget(cfg: Config, dropInfo: DropletInfo): { source: string; owner: string; repo: string } {
  const { ip } = dropInfo; const user = cfg.sshUser; const key = cfg.sshKeyPath;

  // Order: webhookTestRepo > restoreTestRepo > first allow-matched repo from gh api
  for (const cand of [cfg.webhookTestRepo, cfg.restoreTestRepo]) {
    if (!cand) continue;
    const parsed = parseRepoSlug(cand);
    if (!parsed) continue;
    const src = cfg.sources.find((s) => s.name === parsed.owner);
    if (!src) continue;
    if (passesFilter(src, parsed.owner, parsed.repo)) {
      info(`SC#4 target = ${cand} (from config.${cfg.webhookTestRepo === cand ? "webhookTestRepo" : "restoreTestRepo"})`);
      return { source: src.name, owner: parsed.owner, repo: parsed.repo };
    }
  }

  // Discovery fallback: first allow-matched repo from gh api for cfg.sources[0]
  for (const src of cfg.sources) {
    const accountTypeCmd =
      `set -a; source ${REMOTE_ENV(cfg)}; set +a; ` +
      `source ${REMOTE_DETECT_LIB}; detect_account_type ${src.name}`;
    const t = sshCapture(ip, user, key, accountTypeCmd).trim();
    const ep = t === "Organization"
      ? `/orgs/${src.name}/repos?type=all&per_page=100`
      : `/users/${src.name}/repos?type=all&per_page=100`;
    const listCmd =
      `set -a; source ${REMOTE_ENV(cfg)}; set +a; ` +
      `gh api "${ep}" --jq ".[].full_name"`;
    const repos = sshCapture(ip, user, key, listCmd).split("\n").filter((l) => l.length > 0);
    for (const r of repos) {
      const parsed = parseRepoSlug(r);
      if (!parsed) continue;
      if (passesFilter(src, parsed.owner, parsed.repo)) {
        info(`SC#4 target = ${r} (auto-discovered from source "${src.name}")`);
        return { source: src.name, owner: parsed.owner, repo: parsed.repo };
      }
    }
  }
  bail(
    "Phase 7 SC#4 needs at least one whitelisted repo. " +
    "Set config.webhookTestRepo or config.restoreTestRepo to a small repo, " +
    "or add a source with a non-empty allow list."
  );
}
```
